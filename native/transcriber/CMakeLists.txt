cmake_minimum_required(VERSION 3.24)
project(everyhint_transcriber LANGUAGES C CXX)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
FetchContent_Declare(whisper
  GIT_REPOSITORY https://github.com/ggml-org/whisper.cpp.git
  GIT_TAG d1be6fde11ac6e0407606b4e42fe72d34add8037
  GIT_PROGRESS TRUE)
FetchContent_MakeAvailable(whisper)
add_executable(everyhint-transcriber main.cpp)
target_compile_features(everyhint-transcriber PRIVATE cxx_std_17)
target_link_libraries(everyhint-transcriber PRIVATE whisper)
if(WIN32)
  target_compile_definitions(everyhint-transcriber PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(everyhint-transcriber PRIVATE psapi)
endif()
install(TARGETS everyhint-transcriber RUNTIME DESTINATION . COMPONENT EveryHint)
install(FILES ${whisper_SOURCE_DIR}/LICENSE DESTINATION . RENAME whisper-LICENSE COMPONENT EveryHint)
